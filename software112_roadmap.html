RLS-Grundprinzip (kritisch!)

-- Jede Tabelle bekommt tenant_id
ALTER TABLE weinberge ENABLE ROW LEVEL SECURITY;

-- Policy: nur eigener Mandant sichtbar
CREATE POLICY "tenant_select" ON weinberge
  FOR SELECT USING (
    tenant_id IN (SELECT tenant_id
    FROM tenant_memberships
    WHERE user_id = auth.uid())
  );

Auth-Rollen

  • system_admin — ICP Consultants, Zugriff auf alle Mandanten
  • tenant_admin — Weingutbetreiber, verwaltet eigene User
  • winzer — Erfassung Weingarten + Keller
  • leser — Read-Only (Außendienst, Berater)
